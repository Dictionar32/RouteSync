## 2. OUTPUT: CONTRACT LAYER (api-contract.ts)

**Generated by:** `ContractLayer.generate()`  
**From:** manifest.models + manifest.resources

```typescript
// Generated: contract/api-contract.ts

import { z } from 'zod'

// ===== MODEL SCHEMAS (Snake_case dari database)
export const UserSchema = z.object({
  id: z.number(),
  user_name: z.string(),
  user_email: z.string().email(),
  user_phone: z.string().nullable(),
  created_at: z.string().datetime(),
  updated_at: z.string().datetime(),
})
export type User = z.infer<typeof UserSchema>
export const validateUser = (payload: unknown): User => UserSchema.parse(payload)

export const OrderSchema = z.object({
  id: z.number(),
  user_id: z.number(),
  order_status: z.enum(['pending', 'confirmed', 'shipped', 'delivered']),
  total_amount: z.number().decimal(),
  created_at: z.string().datetime(),
  updated_at: z.string().datetime(),
})
export type Order = z.infer<typeof OrderSchema>
export const validateOrder = (payload: unknown): Order => OrderSchema.parse(payload)

// ===== RESOURCE SCHEMAS (dari manifest.resources)
export const UserResourceSchema = z.object({
  id: z.number(),
  user_name: z.string(),
  user_email: z.string(),
  created_at: z.string(),
})
export type UserResource = z.infer<typeof UserResourceSchema>
export const validateUserResource = (payload: unknown): UserResource => UserResourceSchema.parse(payload)

export const OrderResourceSchema = z.object({
  id: z.number(),
  user_id: z.number(),
  order_status: z.string(),
  total_amount: z.number(),
  created_at: z.string(),
})
export type OrderResource = z.infer<typeof OrderResourceSchema>
export const validateOrderResource = (payload: unknown): OrderResource => OrderResourceSchema.parse(payload)

// ===== ROUTE-SPECIFIC SCHEMAS (fallback untuk non-resource routes)
// (Dalam contoh ini semua route adalah resource alias, jadi tidak ada fallback schema)
```

## 4. OUTPUT: SCHEMA LAYER (api-schema.ts)

**Generated by:** `SchemaLayer.generate(routeResponseMap)`  
**From:** manifest.routes dengan schema

```typescript
// Generated: contract/api-schema.ts

import { z } from 'zod'

// ===== FORM VALIDATION SCHEMAS
export const OrderCreatePayloadSchema = z.object({
  customer_id: z.number().int().positive(),
  items: z.array(z.object({
    product_id: z.number().int().positive(),
    quantity: z.number().int().positive().min(1),
  })),
  notes: z.string().optional(),
})
export type OrderCreatePayload = z.infer<typeof OrderCreatePayloadSchema>

// ===== FORM VALUES (untuk react-hook-form)
export const ApiSchema = {
  order: {
    create: OrderCreatePayloadSchema,
  },
}

export type ApiFormValues = OrderCreatePayload

export const ApiDefaultValues: Record<string, Partial<ApiFormValues>> = {
  'order.create': {
    items: [],
    notes: '',
  },
}
```

## 5. OUTPUT: MAPPER LAYER (api-mapper.ts)

**Generated by:** `MapperLayer.generate(routeResponseMap)`  
**From:** manifest.models + routeResponseMap

```typescript
// Generated: mappers/api-mapper.ts

import { User, Order, UserResource, OrderResource } from '../contract/api-contract'
import { UserTransformed, OrderTransformed } from '../types/api-read'
import { OrderCreatePayload } from '../contract/api-schema'
import { OrdersCreateForm } from '../api-form'

// ===== READ MAPPERS (response transform: snake_case → camelCase)
export const toUserRead = (raw: UserResource): UserTransformed => ({
  id: raw.id,
  userName: raw.user_name,
  userEmail: raw.user_email,
  userPhone: raw.user_phone,
  createdAt: raw.created_at,
  updatedAt: raw.updated_at,
})

export const toUserReadList = (raw: UserResource[]): UserTransformed[] =>
  raw.map(toUserRead)

export const toOrderRead = (raw: OrderResource): OrderTransformed => ({
  id: raw.id,
  userId: raw.user_id,
  orderStatus: raw.order_status as OrderTransformed['orderStatus'],
  totalAmount: raw.total_amount,
  createdAt: raw.created_at,
})

export const toOrderReadList = (raw: OrderResource[]): OrderTransformed[] =>
  raw.map(toOrderRead)

// ===== API MAPPERS (form transform: camelCase → snake_case)
export const toApiOrderCreate = (form: OrdersCreateForm['Create']): OrderCreatePayload => ({
  customer_id: form.customerId,
  items: form.items.map(item => ({
    product_id: item.productId,
    quantity: item.quantity,
  })),
  notes: form.notes,
})
```

## 8. ORCHESTRATION FLOW

Urutan eksekusi layer dalam `ZodTierGenerator.generate()`:

```
Input: manifest.json
  │
  ├─→ ContractLayer.generate()
  │   ├─ Emit: UserSchema, OrderSchema, UserResourceSchema, OrderResourceSchema
  │   └─ Return: routeResponseMap
  │       {
  │         'users.show': { zType: 'UserResourceSchema', tsType: 'UserResource', ... },
  │         'users.index': { zType: 'UserResourceSchema', tsType: 'UserResource', collection: true, ... },
  │         'orders.store': { zType: 'OrderSchema', tsType: 'Order', ... },
  │         'orders.index': { zType: 'OrderResourceSchema', tsType: 'OrderResource', collection: true, ... },
  │       }
  │
  ├─→ SchemaLayer.generate(routeResponseMap)
  │   ├─ Read: manifest.routes dengan schema (orders.store)
  │   └─ Emit: OrderCreatePayloadSchema, ApiSchema, ApiDefaultValues
  │
  ├─→ FieldLayer.generate()
  │   ├─ Read: manifest.models fields
  │   └─ Emit: ApiField lookup table
  │
  ├─→ ReadLayer.generate(routeResponseMap)
  │   ├─ Read: manifest.models + routeResponseMap
  │   ├─ Emit: UserTransformed, OrderTransformed interfaces
  │   └─ Emit: UsersShow, UsersIndex, OrdersIndex response types
  │
  ├─→ FormLayer.generate()
  │   ├─ Read: manifest.routes dengan schema
  │   └─ Emit: OrdersForm interface
  │
  └─→ MapperLayer.generate(routeResponseMap)
      ├─ Read: manifest.models + routeResponseMap
      └─ Emit: toUserRead, toOrderRead, toApiOrderCreate mappers

Output: 6 files (contract, schema, field, read, form, mappers)
```

## 9. INTEGRATION DENGAN SDK

Setelah 6 layer selesai, kode lain mengintegrasikan hasil:

```typescript
// api.ts (SDKGenerator)
import { validateUserResource, validateOrderResource } from './contract/api-contract'
import { validateOrderCreatePayload } from './contract/api-schema'
import { toUserRead, toOrderRead, toApiOrderCreate } from './mappers/api-mapper'

export const api = defineApi({
  users: {
    show: {
      route: '/users/{id}',
      method: 'GET',
      contract: validateUserResource,
      mapper: toUserRead,
    },
    index: {
      route: '/users',
      method: 'GET',
      contract: validateUserResource,
      mapper: (raw) => ({
        data: raw.data.map(toUserRead),
        currentPage: raw.currentPage,
        total: raw.total,
      }),
    },
  },
  orders: {
    create: {
      route: '/orders',
      method: 'POST',
      payload: validateOrderCreatePayload,
      transform: toApiOrderCreate,
      contract: validateOrderResource,
      mapper: toOrderRead,
    },
    index: {
      route: '/orders',
      method: 'GET',
      contract: validateOrderResource,
      mapper: (raw) => ({
        data: raw.data.map(toOrderRead),
        currentPage: raw.currentPage,
        total: raw.total,
      }),
    },
  },
})
```

## 10. TYPE SAFETY VERIFICATION

Setelah generate, TypeScript dapat verify:

```typescript
// ✅ Type-safe usage di component
import { useOrders } from '@routesync/api'

const { data } = useOrders.index()
// data: {
//   data: OrderTransformed[]
//   currentPage?: number
//   total?: number
// }

const { mutate: createOrder } = useOrders.create()
await createOrder({
  customerId: 1,        // ✅ number, tidak string
  items: [
    {
      productId: 10,    // ✅ number
      quantity: 2,      // ✅ number, min 1
    }
  ],
  notes: 'Gift wrap',   // ✅ optional string
})

// ❌ TypeScript error jika type salah:
// await createOrder({
//   customerId: 'abc',    // ❌ Type 'string' is not assignable to type 'number'
//   items: [
//     { productId: '10', quantity: 2 }  // ❌ Type 'string' is not assignable to type 'number'
//   ]
// })
```
